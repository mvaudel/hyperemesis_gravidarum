Association with nvp/hgg

association_data <- genotypes %>% 
  inner_join(
    phenotypes %>% 
      select(
        child_sentrix_id,
        all_of(batch_columns),
        sex,
        nvp,
        hg
      ),
    by = "child_sentrix_id"
  )

results <- list()

for (snp in unique(genotypes$id)) {
  
  for (pheno in c("nvp", "hg")) {
  
  variant_association_data <- association_data %>% 
    filter(
      id == snp 
    )
  
  model <- glm(
    formula = as.formula(paste0(pheno, " ~ c + m + sex + ", paste(batch_columns, collapse = " + "))),
    family = "binomial",
    data = variant_association_data
  )
  
  coefs <- summary(model)$coefficients
  
  association_results <- data.frame(
    variant = snp,
    chromosome = variant_association_data$chromosome[1],
    position = variant_association_data$position[1],
    ref = variant_association_data$ref[1],
    alt = variant_association_data$alt[1],
    phenotype = pheno,
    beta_child = coefs["c", "Estimate"],
    se_child = coefs["c", "Std. Error"],
    p_child = coefs["c", "Pr(>|z|)"],
    beta_mother = coefs["m", "Estimate"],
    se_mother = coefs["m", "Std. Error"],
    p_mother = coefs["m", "Pr(>|z|)"],
    n = nrow(variant_association_data)
  )
  
  results[[length(results) + 1]] <- association_results
  
  }
}

results <- do.call(rbind, results)

Align betas on maternal allele

results_aligned <- results

for (i in 1:nrow(results_aligned)) {
  
  if(results_aligned$beta_mother[i] < 0) {
    
    temp_allele <- results_aligned$alt[i]
    results_aligned$alt[i] <- results_aligned$ref[i]
    results_aligned$ref[i] <- temp_allele
    
    results_aligned$beta_mother[i] <- -results_aligned$beta_mother[i]
    results_aligned$beta_child[i] <- -results_aligned$beta_child[i]
    
  }
  
}

Forest plot

snp_order <- results_aligned %>% 
  filter(
    phenotype == "nvp"
  ) %>% 
  arrange(
    desc(p_mother)
  ) %>% 
  pull(
    variant
  )

plot_data <- rbind(
  results_aligned %>% 
    select(
      variant,
      phenotype,
      beta = beta_child,
      se = se_child,
      p = p_child
    ) %>% 
    mutate(
      genome = "child"
    ),
  results_aligned %>% 
    select(
      variant,
      phenotype,
      beta = beta_mother,
      se = se_mother,
      p = p_mother
    ) %>% 
    mutate(
      genome = "mother"
    )
) %>% 
  mutate(
    phenotype_factor = factor(phenotype, levels = c("nvp", "hg")),
    genome_factor = factor(genome, levels = c("child", "mother")),
    variant_factor = factor(variant, levels = snp_order),
    y = as.numeric(variant_factor) + 1/4 * (as.numeric(genome_factor) - 1.5)
  )

levels(plot_data$phenotype_factor) <- c("NVP", "HG")

forest_plot <- ggplot() +
